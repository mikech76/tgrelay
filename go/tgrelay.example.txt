# -----------------------------------------------------------------------------
#  1. PROXIES / ПРОКСИ
# -----------------------------------------------------------------------------
#  EN  HTTP CONNECT proxies that Telegram traffic goes through.
#      One proxy per line:
#
#          PROXY=user:password@host:port
#
#      Without login and password:
#
#          PROXY=host:port
#
#      You can add as many PROXY= lines as you like: the program tries them
#      and picks a working one. More proxies = more stable.
#
#      The type is either set by the scheme or detected by the program:
#
#          PROXY=user:pass@host:port          # detect automatically (HTTP first)
#          PROXY=http://user:pass@host:port   # HTTP CONNECT
#          PROXY=https://user:pass@host:port  # https proxy, TLS as a fallback
#          PROXY=socks5://user:pass@host:port # SOCKS5
#          PROXY=socks://user:pass@host:port  # SOCKS5 as well (5 by default)
#          PROXY=socks4://user:pass@host:port # SOCKS4, socks4a:// for 4a
#
#      Any proxy that allows CONNECT or SOCKS will do, free or paid.
#      Port may be omitted: http 8080, https 443, socks 1080.
#
#      To disable a proxy temporarily, put # at the start of its line.
#      A note after " #" at the end of a line is ignored.
#
#      Examples (copy a line, remove #, put your own data):
#
#      PROXY=user:password@1.2.3.4:9405        # primary
#      PROXY=user:password@5.6.7.8:9829        # backup
#      PROXY=9.9.9.9:8080                      # no auth
#
#      The old one-line format still works:
#      PROXIES=user:pass@1.2.3.4:9405;5.6.7.8:8080
#
#      No PROXY= lines at all = direct mode: the program connects to the
#      internet by itself, without any upstream proxy. That is how a VPS abroad
#      becomes a standalone MTProto + SOCKS5 proxy.
#
#  RU  HTTP CONNECT-прокси, через которые пойдёт трафик Telegram.
#      По одному прокси на строку:
#
#          PROXY=логин:пароль@хост:порт
#
#      Если у прокси нет логина и пароля:
#
#          PROXY=хост:порт
#
#      Строк PROXY= может быть сколько угодно: программа перебирает их
#      и выбирает рабочий. Чем больше прокси, тем стабильнее.
#
#      Тип задаётся схемой, а если схемы нет — определяется сам:
#
#          PROXY=логин:пароль@хост:порт      # определить автоматически (сначала HTTP)
#          PROXY=http://логин:пароль@хост:порт   # HTTP CONNECT
#          PROXY=https://логин:пароль@хост:порт  # https-прокси, при отказе — TLS
#          PROXY=socks5://логин:пароль@хост:порт # SOCKS5
#          PROXY=socks://логин:пароль@хост:порт  # тоже SOCKS5 (по умолчанию 5)
#          PROXY=socks4://логин:пароль@хост:порт # SOCKS4, socks4a:// — 4a
#
#      Подойдёт любой прокси, который умеет CONNECT или SOCKS, платный и бесплатный.
#      Порт можно не писать: http 8080, https 443, socks 1080.
#
#      Чтобы временно отключить прокси, поставьте # в начало строки.
#      Пояснение после " #" в конце строки игнорируется.
#
#      Примеры (скопируйте строку, уберите # и подставьте свои данные):
#
#      PROXY=user:password@1.2.3.4:9405        # основной
#      PROXY=user:password@5.6.7.8:9829        # запасной
#      PROXY=9.9.9.9:8080                      # без авторизации
#
#      Старый формат одной строкой тоже поддерживается:
#      PROXIES=user:pass@1.2.3.4:9405;5.6.7.8:8080
#
#      Ни одной строки PROXY= = режим без прокси (direct): программа выходит
#      в интернет сама, без апстрима. Так обычный VPS за рубежом становится
#      самостоятельным MTProto + SOCKS5-прокси.

PROXY=


# -----------------------------------------------------------------------------
#  2. PORTS / ПОРТЫ
# -----------------------------------------------------------------------------
#  EN  MTProto proxy port. The same port is entered in Telegram when adding
#      an "MTProto" proxy. Default 55055. To disable MTProto completely,
#      comment the line out (#PROXY_PORT=...), leave it empty or set off.
#
#  RU  Порт MTProto-прокси. Этот же порт указывается в Telegram
#      при добавлении прокси типа «MTProto». По умолчанию 55055.
#      Чтобы отключить MTProto совсем, закомментируйте строку
#      (#PROXY_PORT=...), оставьте значение пустым или поставьте off.
PROXY_PORT=55055

#  EN  SOCKS5 proxy port. Entered in Telegram when adding a "SOCKS5" proxy.
#      Default 55056. Disabled the same way: #SOCKS_PORT=..., empty or off.
#
#  RU  Порт SOCKS5-прокси. Указывается в Telegram при добавлении
#      прокси типа «SOCKS5». По умолчанию 55056.
#      Отключается так же: #SOCKS_PORT=... , пусто или off.
SOCKS_PORT=55056

#  EN  Address to listen on for SOCKS5:
#        127.0.0.1  — this computer only (safer);
#        0.0.0.0    — also other devices in the local network.
#
#  RU  На каком адресе слушать SOCKS5:
#        127.0.0.1  — только этот компьютер (безопаснее);
#        0.0.0.0    — ещё и другие устройства в локальной сети.
SOCKS_HOST=0.0.0.0


# -----------------------------------------------------------------------------
#  3. MTProto SECRET / СЕКРЕТ MTProto
# -----------------------------------------------------------------------------
#  EN  The key Telegram uses to encrypt the connection to the MTProto proxy.
#      Exactly 32 characters, 0-9 and a-f (16 bytes).
#
#      If left empty, a key is generated on the first run and written here.
#
#      Generate it by hand:
#          openssl rand -hex 16
#
#      You can put several SECRET= lines: each is a separate key, hand them
#      out to different people and remove the line to revoke it.
#
#      IMPORTANT: if you change the key, the proxy has to be added to
#      Telegram again on every device.
#
#  RU  Ключ, которым Telegram шифрует соединение с MTProto-прокси.
#      Ровно 32 символа из 0-9 и a-f (16 байт).
#
#      Если оставить пустым, ключ сгенерируется при первом запуске и сам
#      запишется сюда.
#
#      Сгенерировать вручную:
#          openssl rand -hex 16
#
#      Строк SECRET= может быть несколько: каждая — отдельный ключ. Раздайте
#      их разным людям, а чтобы отозвать доступ — удалите строку.
#
#      ВАЖНО: если поменять ключ, в Telegram нужно заново добавить прокси
#      на каждом устройстве.
SECRET=

# -----------------------------------------------------------------------------
#  EN  en — English, ru — Russian. Empty = system language.

#
#  RU  en — English, ru — Русский. Пусто = язык системы.
#      Меняется и из меню трея: правый клик -> Язык / Language.
LANG=


# -----------------------------------------------------------------------------
#  4. EXTERNAL NAME / ВНЕШНЕЕ ИМЯ
# -----------------------------------------------------------------------------
#  EN  Only needed if you forwarded the port on your router. DDNS name or
#      external IP of this computer, so the Telegram links use the external
#      address instead of the local one. Example: yourname.duckdns.org
#      If empty, links use the local address of the computer and the proxy
#      works only inside your own network.
#
#  RU  Нужно, только если пробросили порт на роутере. DDNS-имя или внешний
#      IP компьютера — чтобы ссылки для Telegram были с внешним адресом,
#      а не с локальным. Пример: yourname.duckdns.org
#      Если пусто — в ссылках будет локальный адрес компьютера,
#      и прокси будет работать только в своей сети.
EXTERNAL_HOST=


# -----------------------------------------------------------------------------
#  5. SOCKS5 ACCESS / ДОСТУП К SOCKS5
# -----------------------------------------------------------------------------
#  EN  Addresses that may use SOCKS5 without a password. By default: this
#      computer and the local network. Format: networks separated by commas
#      (CIDR). Empty — every address needs the password.
#
#  RU  Адреса, которым SOCKS5 разрешает подключаться без пароля.
#      По умолчанию: сам компьютер и локальная сеть.
#      Формат: список сетей через запятую (CIDR).
#      Пусто — пароль требуется всем.
AUTH_BYPASS=127.0.0.1/32,192.168.0.0/24

#  EN  Login and password for everybody else (for example, connections from
#      the internet). If both are empty, authorization is off and every
#      address is allowed.
#
#      Several SOCKS_USER lines make several accounts; each SOCKS_PASS belongs
#      to the SOCKS_USER line above it. Remove a pair to revoke that account.
#
#      IMPORTANT: before forwarding the port outside, always set a password,
#      otherwise your SOCKS5 becomes an open proxy for the whole internet.
#
#  RU  Логин и пароль для всех остальных (например, для подключения
#      из интернета). Если оба пустые — авторизация выключена, и пускаются
#      все адреса.
#
#      Несколько строк SOCKS_USER — несколько учётных записей; каждая
#      SOCKS_PASS относится к строке SOCKS_USER над ней. Удалите пару, чтобы
#      отозвать доступ.
#
#      ВАЖНО: перед пробросом порта наружу обязательно задайте пароль,
#      иначе ваш SOCKS5 станет открытым прокси для всего интернета.
SOCKS_USER=
SOCKS_PASS=

#
#  RU  Адреса Telegram, по которым проверяется живость прокси из пула.
#      Менять не нужно.
TEST_TARGET=149.154.167.51:443,149.154.175.50:443
